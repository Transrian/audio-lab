# Dependencies of the local model commands (`audiolab serve`, `audiolab train`).
#
# They are NOT in pyproject.toml, so `uv sync`, CI and uv.lock stay light and
# nobody downloads PyTorch to run the evaluation. Install them in their own
# virtual environment (scripts/nemo-venv.sh does exactly this):
#
#   uv venv --python 3.12 .venv-nemo
#   uv pip install --python .venv-nemo/bin/python -e . -r requirements/nemo.txt
#
# Then `.venv-nemo/bin/audiolab serve --model configs/models/<name>.yaml` runs a
# local checkpoint, and `AUDIOLAB_NEMO_PYTHON=$PWD/.venv-nemo/bin/python`
# lets `audiolab eval`, run from the light environment, start that server by
# itself. See docs/stt/training.md for why the pins are separated this way.

# NeMo and the training stack. The versions are the ones verified on the
# training and serving GPUs; NeMo 2.6.1 is the last release that works with
# Python 3.12 and these NumPy/Numba versions.
nemo_toolkit[asr]==2.6.1
torch==2.8.0
torchaudio==2.8.0
lightning==2.4.0
torchmetrics==1.6.1

# The TDT loss is a Numba CUDA kernel, compiled at the first training step: it
# needs these versions and a CUDA toolkit, which is why training runs inside the
# CUDA devel image the container script names.
numba==0.61.2
numba-cuda==0.15.1
cuda-python==12.8.0

# PyTorch 2.8.0 built for CUDA 12.8, the build the GPUs need.
--extra-index-url https://download.pytorch.org/whl/cu128
